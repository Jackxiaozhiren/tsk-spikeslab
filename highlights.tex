% Highlights for: Sparse Bayesian TSK with Spike-and-Slab Priors
% Target: Applied Soft Computing (Elsevier)
% Format: 3-5 bullets, each <=85 characters

\begin{highlights}
\item First spike-and-slab prior for TSK fuzzy rule consequent selection.
\item Analytical BIC-based approximation for fast rule selection without MCMC.
\item 38\% (3.1/5) rule reduction on Energy benchmarks; 0\% on Concrete.
\item Non-sparse Bayesian TSK achieves well-calibrated intervals (PICP 0.90--0.96).
\item Sparsity-accuracy-calibration trade-off characterized across three benchmarks.
\end{highlights}
